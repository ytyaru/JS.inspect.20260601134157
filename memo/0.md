# inspect

対象の文字列表現を出力する。

```javascript
import {inspect} of './inspect.js';
```
```javascript
<script src="./inspect.js"></script>
```
```javascript
inspect(v); // vの内容を意味する文字列を返す
```
```javascript
inspect(v,options);
```
```javascript
const options = {
    str: {
        quote: `'`,
    },
    sym: {
        quote: `'`,
        mode: `standard`,
    },
    ins: {
        mode: `lost`,
    },
    obj: {
        indent: 0, // k:vのインデント 0,1,2,4,8
        space: 0, // k: vの半角スペース 0,1
    },
};
```
